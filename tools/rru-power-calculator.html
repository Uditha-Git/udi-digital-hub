// ===================================== // MESSAGE FUNCTION // ===================================== function showMessage(text,color){ let msg=document.getElementById("message"); msg.innerText=text; msg.style.color=color; msg.style.display="block"; setTimeout(()=>{ msg.style.display="none"; },2500); } // ===================================== // WATT TO dBm // ===================================== function wattToDbm(watt){ if(watt<=0){ return 0; } return 10 * Math.log10(watt * 1000); } // ===================================== // MAIN CALCULATION // ===================================== function calculatePower(){ let rruPower = Number(document.getElementById("rruPower").value); let trx = Number(document.getElementById("trxMode").value); let cells = Number(document.getElementById("cells").value); let carriers = Number(document.getElementById("carriers").value); let rsPower = Number(document.getElementById("rsPower").value); let maxRsPower = Number(document.getElementById("maxRsPower").value); let pa = Number(document.getElementById("paValue").value); let pb = Number(document.getElementById("pbValue").value); // ===================================== // VALIDATION // ===================================== if( rruPower<=0 || trx<=0 || cells<=0 || carriers<=0 ){ showMessage( "❌ Please enter valid values", "red" ); return; } if(rsPower > maxRsPower){ showMessage( "❌ RS Power exceeds maximum allowed level", "red" ); return; } // ===================================== // POWER CALCULATIONS // ===================================== // Power per TX chain let txPower = rruPower / trx; // Power per cell let cellPower = rruPower / cells; // Power per carrier let carrierPower = rruPower / (cells * carriers); // dBm conversion let totalDbm = wattToDbm(rruPower); let txDbm = wattToDbm(txPower); // ===================================== // RS POWER ANALYSIS // ===================================== let remainingIncrease = maxRsPower - rsPower; let rsRecommendation; if(remainingIncrease > 0){ rsRecommendation = "✅ Can increase RS Power by " + remainingIncrease.toFixed(1) + " dB"; } else{ rsRecommendation = "❌ Maximum RS Power reached"; } // ===================================== // RADIO MODE NAME // ===================================== let trxName; switch(trx){ case 2: trxName="2T2R"; break; case 4: trxName="4T4R"; break; case 8: trxName="8T8R"; break; case 16: trxName="16T16R"; break; case 32: trxName="32T32R"; break; case 64: trxName="64T64R"; break; default: trxName= trx+"T"+trx+"R"; } // ===================================== // DISPLAY RESULTS // ===================================== document.getElementById("totalPower").innerText = rruPower.toFixed(2)+" W"; document.getElementById("trxResult").innerText = trxName; document.getElementById("txPower").innerText = txPower.toFixed(2)+" W"; document.getElementById("cellPower").innerText = cellPower.toFixed(2)+" W"; document.getElementById("carrierPower").innerText = carrierPower.toFixed(2)+" W"; document.getElementById("totalDbm").innerText = totalDbm.toFixed(2)+" dBm"; document.getElementById("txDbm").innerText = txDbm.toFixed(2)+" dBm"; // RS / PA / PB document.getElementById("rsResult").innerText = rsPower.toFixed(1)+" dBm"; document.getElementById("maxRsResult").innerText = maxRsPower.toFixed(1)+" dBm"; document.getElementById("paResult").innerText = pa.toFixed(1)+" dB"; document.getElementById("pbResult").innerText = pb.toFixed(1)+" dB"; document.getElementById("increaseResult").innerText = remainingIncrease.toFixed(1)+" dB"; document.getElementById("recommendation").innerText = rsRecommendation; showMessage( "✅ Calculation Completed", "green" ); } // ===================================== // COPY RESULT // ===================================== function copyResult(){ if(document.getElementById("totalPower").innerText=="-"){ showMessage( "❌ Please calculate first", "red" ); return; } let result = `RRU Power Calculator V3 Total RRU Power: ${document.getElementById("totalPower").innerText} Radio Configuration: ${document.getElementById("trxResult").innerText} TX Chain Power: ${document.getElementById("txPower").innerText} Cell Power: ${document.getElementById("cellPower").innerText} Carrier Power: ${document.getElementById("carrierPower").innerText} Total Output Power: ${document.getElementById("totalDbm").innerText} TX Chain dBm: ${document.getElementById("txDbm").innerText} Reference Signal Power: ${document.getElementById("rsResult").innerText} Maximum RS Power: ${document.getElementById("maxRsResult").innerText} PA Value: ${document.getElementById("paResult").innerText} PB Value: ${document.getElementById("pbResult").innerText} Available RS Increase: ${document.getElementById("increaseResult").innerText} Recommendation: ${document.getElementById("recommendation").innerText} `; navigator.clipboard.writeText(result); showMessage( "📋 Result copied successfully", "green" ); } // ===================================== // EXPORT CSV // ===================================== function exportCSV(){ if(document.getElementById("totalPower").innerText=="-"){ showMessage( "❌ Please calculate first", "red" ); return; } let csv = `Parameter,Value Total RRU Power,${document.getElementById("totalPower").innerText} Radio Configuration,${document.getElementById("trxResult").innerText} TX Chain Power,${document.getElementById("txPower").innerText} Cell Power,${document.getElementById("cellPower").innerText} Carrier Power,${document.getElementById("carrierPower").innerText} Total Output dBm,${document.getElementById("totalDbm").innerText} TX Chain dBm,${document.getElementById("txDbm").innerText} Reference Signal Power,${document.getElementById("rsResult").innerText} Maximum RS Power,${document.getElementById("maxRsResult").innerText} PA Value,${document.getElementById("paResult").innerText} PB Value,${document.getElementById("pbResult").innerText} Available RS Increase,${document.getElementById("increaseResult").innerText} Recommendation,${document.getElementById("recommendation").innerText}`; let blob = new Blob( [csv], { type:"text/csv" } ); let link=document.createElement("a"); link.href=URL.createObjectURL(blob); link.download="rru-power-calculator-v3.csv"; link.click(); showMessage( "📄 CSV exported successfully", "green" ); } // ===================================== // CLEAR DATA // ===================================== function clearData(){ document.getElementById("rruPower").value=""; document.getElementById("trxMode").value="4"; document.getElementById("cells").value=""; document.getElementById("carriers").value=""; document.getElementById("rsPower").value="18.2"; document.getElementById("maxRsPower").value="20.2"; document.getElementById("paValue").value="0"; document.getElementById("pbValue").value="-3"; document.getElementById("totalPower").innerText="-"; document.getElementById("trxResult").innerText="-"; document.getElementById("txPower").innerText="-"; document.getElementById("cellPower").innerText="-"; document.getElementById("carrierPower").innerText="-"; document.getElementById("totalDbm").innerText="-"; document.getElementById("txDbm").innerText="-"; document.getElementById("rsResult").innerText="-"; document.getElementById("maxRsResult").innerText="-"; document.getElementById("paResult").innerText="-"; document.getElementById("pbResult").innerText="-"; document.getElementById("increaseResult").innerText="-"; document.getElementById("recommendation").innerText="-"; document.getElementById("message").style.display="none"; }
